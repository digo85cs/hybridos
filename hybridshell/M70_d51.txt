                            if c.as_str().eq_ignore_ascii_case("k") {
                                // M64: tema ciclico escuro -> claro -> laranja -> escuro
                                let atual = tema_carregar();
                                let proximo = if atual == 0 { 1 } else if atual == 1 { 2 } else { 0 };
                                tema_salvar(proximo);
                                let nome = if proximo == 1 { "CLARO" } else if proximo == 2 { "LARANJA" } else { "ESCURO" };
                                self.notif.disparar(self.frame, "TEMA", &format!("{} ATIVADO", nome));
                                self.draw();
                                return;
                            }
