// M67: cor de destaque personalizada (hybridos_accent.txt = hex RGB; padrao verde)
const ACCENT_PATH: &str = "C:/Users/Rodrigo/hybridos_accent.txt";

fn accent_carregar() -> u32 {
    if let Ok(txt) = std::fs::read_to_string(ACCENT_PATH) {
        let t = txt.trim();
        if t.len() >= 6 {
            if let Ok(v) = u32::from_str_radix(t, 16) {
                return 0xFF000000 | v;
            }
        }
    }
    0xFFA6E3A1
}
