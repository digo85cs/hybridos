                if self.historico.is_empty() {
                    self.print("(historico vazio)");

                    let copia: Vec<String> = self.historico.clone();
                    for (i86, h86) in copia.iter().enumerate() {
                        self.print(&format!("  {:>3} {}", i86 + 1, h86));
                    }
